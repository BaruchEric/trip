**Tech Stack:** TypeScript strict, Bun, `bun:test`, `@libsql/client`.

import { test, expect, describe } from "bun:test";
import { calibrate } from "@/calibrate";
import type { MeasuredLeg } from "@/legs";

const TESTBED = { lat: 29.5537638, lon: 106.5368476 };
const LIZIBA = { lat: 29.5556826, longitude: 106.5338753, lon: 106.5338753 };
const HONGYA = { lat: 29.5650738, lon: 106.5753425 };

function leg(
  from: { lat: number; lon: number }, to: { lat: number; lon: number },
  minutes: number, source = "osrm-foot", mode = "walking",
): MeasuredLeg {
  return {
    fromLat: from.lat, fromLon: from.lon, toLat: to.lat, toLon: to.lon,
    mode, source, minutes, meters: 1000, fetchedAt: "2026-07-27T12:00:00Z",
  };
}

describe("calibrate", () => {
  test("no legs is UNKNOWN, never agreement", () => {
    // Absence is loud: no measurement is not "the model is perfect here".
    const c = calibrate([], "walking");
    expect(c.legCount).toBe(0);
    expect(c.bands.every((b) => b.medianRatio === null)).toBe(true);
    expect(c.worst).toBeNull();
  });

  test("compares against the MIDPOINT of the two routers, not the slower", () => {
    // D2. The schedule reads the max because that is a scheduling policy;
    // calibration asks what the routers measured, which is the midpoint.
    // Model says 6 for this pair.
    const c = calibrate([
      leg(TESTBED, LIZIBA, 20, "osrm-foot"),
      leg(TESTBED, LIZIBA, 30, "valhalla-pedestrian"),
    ], "walking");
    // 6 / 25 = 24%, not 6 / 30 = 20%.
    expect(c.bands[0]!.medianRatio).toBeCloseTo(6 / 25, 3);
  });

  test("a single source is used alone", () => {
    const c = calibrate([leg(TESTBED, LIZIBA, 22, "osrm-foot")], "walking");
    expect(c.bands[0]!.medianRatio).toBeCloseTo(6 / 22, 3);
  });

  test("bands split on STRAIGHT-LINE distance at 2 km", () => {
    // Not on measured time -- that would band by the quantity under
    // measurement.
    const c = calibrate([
      leg(TESTBED, LIZIBA, 22),      // 0.36 km straight
      leg(TESTBED, HONGYA, 71),      // 3.9 km straight
    ], "walking");
    expect(c.bands[0]!.label).toContain("2 km");
    expect(c.bands[0]!.legCount).toBe(1);
    expect(c.bands[1]!.legCount).toBe(1);
  });

  test("an empty band is null, never 100%", () => {
    const c = calibrate([leg(TESTBED, LIZIBA, 22)], "walking");
    expect(c.bands[1]!.legCount).toBe(0);
    expect(c.bands[1]!.medianRatio).toBeNull();
  });

  test("uses the MEDIAN, so one outlier cannot move it", () => {
    const legs = [
      leg(TESTBED, LIZIBA, 22), leg(LIZIBA, TESTBED, 22),
      leg(TESTBED, LIZIBA, 2000, "valhalla-pedestrian"),
    ];
    const c = calibrate(legs, "walking");
    expect(c.bands[0]!.medianRatio!).toBeGreaterThan(0.005);
  });

  test("names the single worst leg by ratio", () => {
    const c = calibrate([
      leg(TESTBED, LIZIBA, 22),
      leg(TESTBED, HONGYA, 71),
    ], "walking");
    expect(c.worst!.modelMinutes).toBe(6);
    expect(c.worst!.measuredMinutes).toBe(22);
  });

  test("legs of another mode are ignored", () => {
    expect(calibrate([leg(TESTBED, LIZIBA, 22, "osrm-foot", "transit")], "walking").legCount)
      .toBe(0);
  });

  test("a ratio ABOVE 1 means the model is pessimistic", () => {
    // Amsterdam's case, and the one the Chongqing-only version of this
    // project would never have produced.
    const c = calibrate([leg(TESTBED, LIZIBA, 3)], "walking");
    expect(c.bands[0]!.medianRatio!).toBeGreaterThan(1);
  });
});

```ts
import { travelMinutes, haversineKm } from "@/plan/geo";
import { legKey, type MeasuredLeg } from "@/legs";
import type { Mode, Point } from "@/plan/types";

/** How wrong `geo.ts` is, here, measured against legs already stored.
 *
 *  A DERIVATION, not a measurement: `trip route` did the measuring. This file
 *  runs offline and answers the question M9's recon raised -- the model is
 *  excellent in Amsterdam and catastrophic in Chongqing, and nothing in the
 *  tool knew which city it was in. */

export interface LegComparison {
  from: Point;
  to: Point;
  straightKm: number;
  modelMinutes: number;
  /** Midpoint across sources (D2), NOT the maximum the schedule reads. */
  measuredMinutes: number;
  sources: number;
  ratio: number;
}

export interface Band {
  label: string;
  legCount: number;
  /** null means NO LEGS IN THIS BAND. Never 1, which would read as
   *  "the model agrees exactly". */
  medianRatio: number | null;
}

export interface Calibration {
  mode: Mode;
  legCount: number;
  bands: Band[];
  worst: LegComparison | null;
}

/** The band edge, in straight-line km. The finding is that error concentrates
 *  on SHORT hops, so the band must be drawn on distance and not on the
 *  measured time it is being compared against. */
export const BAND_EDGE_KM = 2;

export function calibrate(legs: MeasuredLeg[], mode: Mode): Calibration {
  // Group by directed key so the two routers meet before anything is compared.
  const byLeg = new Map<string, MeasuredLeg[]>();
  for (const l of legs) {
    if (l.mode !== mode) continue;
    const k = legKey(l.fromLat, l.fromLon, l.toLat, l.toLon, l.mode);
    const at = byLeg.get(k);
    if (at) at.push(l);
    else byLeg.set(k, [l]);
  }

  const comparisons: LegComparison[] = [];
  for (const group of byLeg.values()) {
    const first = group[0]!;
    const from = { latitude: first.fromLat, longitude: first.fromLon };
    const to = { latitude: first.toLat, longitude: first.toLon };
    // Midpoint, not max: D2. The max is a scheduling policy that buys safety
    // margin, and folding it in here would report the model as more wrong
    // than it is.
    const measured =
      group.reduce((n, l) => n + l.minutes, 0) / group.length;
    // A zero-length measurement would divide by zero and produce Infinity,
    // which is not a ratio. Such a leg says nothing and is dropped.
    if (measured <= 0) continue;
    const model = travelMinutes(from, to, mode);
    comparisons.push({
      from, to, straightKm: haversineKm(from, to),
      modelMinutes: model, measuredMinutes: measured,
      sources: group.length, ratio: model / measured,
    });
  }

  const under = comparisons.filter((c) => c.straightKm < BAND_EDGE_KM);
  const over = comparisons.filter((c) => c.straightKm >= BAND_EDGE_KM);

  return {
    mode,
    legCount: comparisons.length,
    bands: [
      band(`under ${BAND_EDGE_KM} km`, under),
      band(`${BAND_EDGE_KM} km and over`, over),
    ],
    // Worst = furthest from 1 in the OPTIMISTIC direction where one exists,
    // else furthest from 1 either way. An under-estimate is what makes a day
    // run late, so it is what the user feels.
    worst: worstOf(comparisons),
  };
}

function band(label: string, cs: LegComparison[]): Band {
  return {
    label,
    legCount: cs.length,
    medianRatio: cs.length === 0 ? null : median(cs.map((c) => c.ratio)),
  };
}

/** Median, never mean: one 4.64-detour outlier would drag a mean and
 *  misdescribe the typical hop. */
function median(xs: number[]): number {
  const s = [...xs].sort((a, b) => a - b);
  const mid = Math.floor(s.length / 2);
  return s.length % 2 === 1 ? s[mid]! : (s[mid - 1]! + s[mid]!) / 2;
}

function worstOf(cs: LegComparison[]): LegComparison | null {
  if (cs.length === 0) return null;
  const optimistic = cs.filter((c) => c.ratio < 1);
  const pool = optimistic.length > 0 ? optimistic : cs;
  return pool.reduce((w, c) =>
    Math.abs(Math.log(c.ratio)) > Math.abs(Math.log(w.ratio)) ? c : w);
}
```
